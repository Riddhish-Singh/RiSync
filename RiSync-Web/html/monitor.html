code How to log activity from other pages

Add this small helper to any RiSync page (or put it in a shared JS file):

// RiSync Activity Logger
function RiSyncLog(module, action, details = {}) {
  try {
    const key = 'risync_activity';
    const log = JSON.parse(localStorage.getItem(key) || '[]');
    log.unshift({
      id: crypto.randomUUID ? crypto.randomUUID() : Date.now() + Math.random(),
      ts: new Date().toISOString(),
      module,
      action,
      details,
      page: location.pathname.split('/').pop() || 'unknown'
    });
    if (log.length > 800) log.length = 800;
    localStorage.setItem(key, JSON.stringify(log));
    window.dispatchEvent(new CustomEvent('risync-activity'));
  } catch (e) { console.warn('RiSyncLog failed', e); }
}

// Example usage:
RiSyncLog('Calculator', 'Computed expression', { expr: '2+2', result: 4 });
RiSyncLog('Study Timer', 'Session completed', { duration: '25m', mode: 'Pomodoro' });
RiSyncLog('NCERT', 'Opened chapter', { subject: 'Physics', chapter: 'Motion' });

The monitor listens to both storage events and a custom risync-activity event, so it updates instantly even in the same tab.
